## 概述

本文档描述前端需要实现的交互方法与数据流。

前端通过 WebSocket 与后端通信，负责：
- 用户登录与房间管理
- 收集用户输入（操作）
- 发送 JSON 消息给后端
- 接收后端推送的 `game_state` / `hand_result` / `error`
- 根据数据更新 DOM 渲染

## 四、数据库 (SQLite)

### 4.1 概述

项目使用 SQLite 作为轻量级数据库，无需安装独立的数据库服务。数据库文件为 `server/poker.db`，由 `app.py` 启动时自动初始化。

### 4.2 数据库表结构

#### users 表

```sql
CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT UNIQUE NOT NULL,
    password TEXT NOT NULL,
    chips INTEGER DEFAULT 1000,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

#### rooms 表

```sql
CREATE TABLE IF NOT EXISTS rooms (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    max_players INTEGER DEFAULT 8,
    initial_chips INTEGER DEFAULT 1000,
    created_by INTEGER,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (created_by) REFERENCES users(id)
);
```

#### game_logs 表

```sql
CREATE TABLE IF NOT EXISTS game_logs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    room_id INTEGER,
    player_id INTEGER,
    action TEXT,
    amount INTEGER DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (room_id) REFERENCES rooms(id),
    FOREIGN KEY (player_id) REFERENCES users(id)
);
```

### 4.3 字段说明

**users 表**

| 字段 | 类型 | 说明 |
|------|------|------|
| id | INTEGER | 用户唯一标识，自增 |
| username | TEXT | 用户名，唯一约束，2~20个字符 |
| password | TEXT | 密码（当前明文存储） |
| chips | INTEGER | 筹码余额，默认 1000 |
| created_at | TIMESTAMP | 创建时间 |

**rooms 表**

| 字段 | 类型 | 说明 |
|------|------|------|
| id | INTEGER | 房间唯一标识，自增 |
| name | TEXT | 房间名称 |
| max_players | INTEGER | 最大玩家数，默认 8 |
| initial_chips | INTEGER | 初始筹码，默认 1000 |
| created_by | INTEGER | 创建者用户ID（外键） |
| created_at | TIMESTAMP | 创建时间 |

**game_logs 表**

| 字段 | 类型 | 说明 |
|------|------|------|
| id | INTEGER | 日志唯一标识，自增 |
| room_id | INTEGER | 房间ID（外键） |
| player_id | INTEGER | 玩家ID（外键） |
| action | TEXT | 操作类型 |
| amount | INTEGER | 操作金额，默认 0 |
| created_at | TIMESTAMP | 操作时间 |

### 4.5 Python 连接示例

```python
import sqlite3
import os

DB_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "poker.db")

def get_db():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn

# 用户注册
def register(username, password):
    if len(username) < 2 or len(username) > 20:
        return False, '用户名需要2~20个字符'
    if not password:
        return False, '密码不能为空'
    
    conn = get_db()
    try:
        cursor = conn.cursor()
        cursor.execute('INSERT INTO users (username, password) VALUES (?, ?)', 
                      (username, password))
        conn.commit()
        return True, '注册成功'
    except sqlite3.IntegrityError:
        return False, '用户名已存在'
    finally:
        conn.close()

# 用户登录
def login(username, password):
    conn = get_db()
    try:
        cursor = conn.cursor()
        cursor.execute('SELECT id, username, chips FROM users WHERE username = ? AND password = ?',
                       (username, password))
        user = cursor.fetchone()
        if user:
            return True, '登录成功', user['id'], user['chips']
        return False, '用户名或密码错误', None, None
    finally:
        conn.close()
```

## 五、用户名输入交互流程

### 5.1 为什么需要用户名输入？

默认情况下，后端给每个玩家分配名称 `"玩家1"` ~ `"玩家8"`。为了让玩家能自定义显示名称（如 `"小明"`、`"Alice"`），需要在加入游戏时传入 `name` 字段。

### 2.4 加入逻辑（JS 实现）

```javascript
// ========== ui.js ==========

let ws = null;           // WebSocket 连接对象
let mySeat = null;       // 本客户端的座位号
let myName = "";         // 本客户端的用户名

/**
 * 点击"加入游戏"按钮
 */
function onJoinClick() {
    const nameInput = document.getElementById("input-name");
    const seatSelect = document.getElementById("input-seat");
    const errorEl = document.getElementById("join-error");

    // 1. 校验用户名
    const rawName = nameInput.value.trim();
    if (rawName.length === 0) {
        errorEl.textContent = "用户名不能为空";
        errorEl.style.display = "block";
        return;
    }
    if (rawName.length > 12) {
        errorEl.textContent = "用户名最多 12 个字符";
        errorEl.style.display = "block";
        return;
    }

    // 2. 获取座位号
    const seat = parseInt(seatSelect.value);

    // 3. 保存本地状态
    myName = rawName;
    mySeat = seat;

    // 4. 连接 WebSocket（如果还没连接）
    if (!ws || ws.readyState !== WebSocket.OPEN) {
        connectWebSocket();
    }

    // 5. 发送 join 消息
    sendJoin(seat, rawName);

    // 6. 隐藏错误提示
    errorEl.style.display = "none";
}

/**
 * 发送 join 消息
 */
function sendJoin(seat, name) {
    const msg = {
        type: "join",
        data: {
            seat: seat,
            name: name
        }
    };
    ws.send(JSON.stringify(msg));
    console.log("[发送] join:", msg);
}

// 绑定按钮事件
document.getElementById("btn-join").addEventListener("click", onJoinClick);

// 支持回车键提交
document.getElementById("input-name").addEventListener("keydown", (e) => {
    if (e.key === "Enter") {
        onJoinClick();
    }
});
```

## 五、WebSocket 连接与消息分发

### 3.1 连接管理

```javascript
// ========== app.js ==========

const WS_URL = "ws://127.0.0.1:9002";

function connectWebSocket() {
    ws = new WebSocket(WS_URL);

    ws.onopen = () => {
        console.log("[WebSocket] 已连接");
    };

    ws.onmessage = (event) => {
        const msg = JSON.parse(event.data);
        handleMessage(msg);
    };

    ws.onclose = () => {
        console.log("[WebSocket] 连接断开，3 秒后重连...");
        setTimeout(connectWebSocket, 3000);
    };

    ws.onerror = (err) => {
        console.error("[WebSocket] 错误:", err);
    };
}

/**
 * 消息分发中心
 * 根据 msg.type 路由到不同的处理函数
 */
function handleMessage(msg) {
    console.log("[收到]", msg.type, msg);

    switch (msg.type) {
        case "game_state":
            handleGameState(msg.data);
            break;

        case "hand_result":
            handleHandResult(msg.data);
            break;

        case "error":
            handleError(msg.data);
            break;

        default:
            console.warn("[未知消息类型]", msg.type);
    }
}
```

### 3.2 各消息类型的处理

#### `game_state` — 游戏状态全量同步

```javascript
function handleGameState(data) {
    // 1. 更新阶段显示
    updatePhase(data.phase);

    // 2. 更新公共牌
    updateCommunityCards(data.communityCards);

    // 3. 更新每个玩家
    data.players.forEach((player, index) => {
        updatePlayerRow(index, player);
    });

    // 4. 更新底池
    updatePot(data.pot);

    // 5. 更新当前下注额
    updateCurrentBet(data.currentBet);

    // 6. 更新庄家标记
    updateDealer(data.dealerIndex);

    // 7. 根据是否轮到本客户端，启用/禁用操作按钮
    const isMyTurn = data.players.find(p => p.id === myUserId)?.isCurrent;
    setActionButtonsEnabled(isMyTurn);

    // 8. 更新本客户端的筹码显示
    if (myUserId) {
        const me = data.players.find(p => p.id === myUserId);
        if (me) {
            document.getElementById("remaining-chips").textContent = me.chips;
            document.getElementById("current-chips").textContent = me.currentBet;
        }
    }
}
```

#### `hand_result` — 摊牌结果

```javascript
function handleHandResult(data) {
    // 显示赢家信息
    const winnerNames = data.winners.map(w => {

    // 可选：高亮赢家行

}
```

## 六、DOM 渲染映射

### 6.1 玩家行更新

每个玩家在 HTML 中对应一个 `.player-row`，需要动态更新以下内容：

```javascript
/**
 * 更新单个玩家的 DOM 显示
 * @param {number} index - 玩家索引 0~7
 * @param {object} player - Player JSON 对象
 */
function updatePlayerRow(index, player) {
    const row = document.querySelectorAll(".player-row")[index];
    if (!row) return;

    // 1. 更新玩家名称（含自定义名称）
    const nameEl = row.querySelector(".player-name");
    if (nameEl) {
        nameEl.textContent = player.name;  // ← 这里显示自定义名称
    }

    // 2. 更新动作文本
    const actionEl = row.querySelector(`#p${index + 1}-action`);
    if (actionEl) {
        actionEl.textContent = getActionText(player.action);
    }

    // 3. 更新状态文本
    const stateEl = row.querySelector(`#p${index + 1}-state`);
    if (stateEl) {
        stateEl.textContent = getStateText(player.state);
    }

    // 4. 更新 CSS 类（高亮当前玩家、标记弃牌/出局）
    row.classList.remove("active", "folded", "out");
    if (player.isCurrent) {
        row.classList.add("active");
    }
    if (player.state === "folded") {
        row.classList.add("folded");
    }
    if (player.state === "out") {
        row.classList.add("out");
    }
}

function getActionText(action) {
    const map = {
        "": "等待",
        "fold": "弃牌",
        "check": "让牌",
        "call": "跟注",
        "raise": "加注"
    };
    return map[action] || action;
}

function getStateText(state) {
    const map = {
        "active": "存活",
        "folded": "弃牌",
        "out": "出局"
    };
    return map[state] || state;
}
```

## 七、玩家操作发送

```javascript
// ========== events.js ==========

/**
 * 发送玩家操作
 * @param {string} action - "fold" | "check" | "call" | "raise"
 * @param {number} amount - 仅 raise 时需要
 */
function sendAction(action, amount = 0) {
    if (!ws || ws.readyState !== WebSocket.OPEN) {
        console.error("[错误] WebSocket 未连接");
        return;
    }

    const msg = {
        type: "player_action",
        data: {
            action: action
        }
    };

    if (action === "raise") {
        msg.data.amount = amount;
    }

    ws.send(JSON.stringify(msg));
    console.log("[发送] player_action:", msg);
}

// 绑定按钮事件

    const amount = prompt("输入加注金额:");
    if (amount && !isNaN(amount)) {
        sendAction("raise", parseInt(amount));
    }
});

document.querySelector(".btn:nth-child(4)").addEventListener("click", () => sendAction("fold"));
```

### 7.1 前端 → 后端

```json

    "type": "join",
```

|------|------|------|------|

## 十二、与后端的约定

| 约定 | 说明 |
|------|------|
| 用户名长度 | 前端限制 1~12 字符，后端也会 `strip()[:12]` |
| 名称跨局保持 | `reset_for_new_hand()` 不重置 `name`，一局结束后名称不变 |
| 断线重连 | 重连后需重新发送 `join`（含 `name`），后端会覆盖旧名称 |
| 座位冲突 | 同一座位号后连接的客户端会覆盖先连接的（`connections[seat] = ws`） |
| 名称唯一性 | 不强制唯一，允许多个玩家同名 |
